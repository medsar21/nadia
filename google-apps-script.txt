const SHEET_ID = '1qoB9iVUOnD45OBlWjJxhQ5s6L4IZKK1B6GmyJxgqoag';
const SHEET_NAME = 'nadia';

/**
 * Handles POST from form (application/x-www-form-urlencoded or multipart/form-data)
 * or JSON (application/json). Returns JSON via ContentService (no custom headers).
 */
function doPost(e) {
  if (!e) {
    Logger.log('doPost called with undefined event');
    return jsonResponse({ status: 'error', message: 'No event object received' });
  }

  try {
    let data = {};

    // Form POST: e.parameter may contain string values or arrays
    if (e.parameter && Object.keys(e.parameter).length) {
      data = {
        fullName: firstString(e.parameter.fullName),
        phone: firstString(e.parameter.phone),
        email: firstString(e.parameter.email),
        address: firstString(e.parameter.address),
        packChoice: firstString(e.parameter.packChoice)
      };
    } else if (e.postData && e.postData.contents) {
      // JSON body
      try {
        data = JSON.parse(e.postData.contents);
      } catch (err) {
        return jsonResponse({ status: 'error', message: 'Invalid JSON: ' + err.toString() });
      }
    } else {
      return jsonResponse({ status: 'error', message: 'No data received' });
    }

    // Minimal validation
    if (!data.fullName || !data.phone || !data.email || !data.packChoice) {
      return jsonResponse({ status: 'error', message: 'Missing required fields: fullName, phone, email, packChoice' });
    }

    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = createSheet(ss);
    }

    // If sheet is new or empty, ensure headers exist
    if (sheet.getLastRow() === 0) {
      addHeaders(sheet);
    }

    const timestamp = new Date();
    const row = [
      timestamp,
      data.fullName,
      data.phone,
      data.email,
      data.address || '',
      data.packChoice
    ];

    sheet.appendRow(row);
    Logger.log('Row appended. Last row: ' + sheet.getLastRow());

    return jsonResponse({ status: 'success', message: 'Saved', row: sheet.getLastRow() });
  } catch (err) {
    Logger.log('doPost error: ' + err.toString());
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function doGet(e) {
  return jsonResponse({ status: 'ok', message: 'Web App running' });
}

function doOptions(e) {
  // Keep simple: ContentService does not support setting custom response headers.
  return ContentService.createTextOutput('').setMimeType(ContentService.MimeType.TEXT);
}

/* Helpers */

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function firstString(val) {
  if (Array.isArray(val)) return String(val[0] || '').trim();
  return String(val || '').trim();
}

function createSheet(ss) {
  const sheet = ss.insertSheet(SHEET_NAME);
  addHeaders(sheet);
  return sheet;
}

function addHeaders(sheet) {
  const headers = ['Date/Heure', 'Nom complet', 'Téléphone', 'Email', 'Adresse', 'Pack choisi'];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  const headerRange = sheet.getRange(1, 1, 1, headers.length);
  headerRange.setFontWeight('bold');
  headerRange.setBackground('#6B8FA3');
  headerRange.setFontColor('#FFFFFF');
  headerRange.setHorizontalAlignment('center');

  sheet.setColumnWidth(1, 180);
  sheet.setColumnWidth(2, 200);
  sheet.setColumnWidth(3, 150);
  sheet.setColumnWidth(4, 250);
  sheet.setColumnWidth(5, 200);
  sheet.setColumnWidth(6, 200);
  sheet.setFrozenRows(1);
}
